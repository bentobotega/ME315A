process_chunk <- function(chunk, pos) {
  chunk |>
    drop_na() |>
    filter(!str_starts(DESTINATION_AIRPORT, "1")) |>
    group_by(DESTINATION_AIRPORT) |>
    summarize(
      chunk_delay = sum(ARRIVAL_DELAY),
      chunk_n = n(),
      .groups = "drop"
    )
}

chunks <- read_csv_chunked(
  file = "../datasets/flights.csv",
  callback = DataFrameCallback$new(process_chunk),
  chunk_size = 1000000,
  col_types = cols_only(
    DESTINATION_AIRPORT = col_character(),
    ARRIVAL_DELAY = col_double()
  )
)

flights <- chunks |>
  group_by(DESTINATION_AIRPORT) |>
  summarize(
    mean_delay = sum(chunk_delay)/sum(chunk_n),
    .groups = "drop"
  )
